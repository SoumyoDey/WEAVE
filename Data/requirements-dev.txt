# Dev/test-only dependencies (not needed to run the API in production).
#   pip install -r requirements.txt -r requirements-dev.txt
#
# Run the backend tests:
#   cd Data && python -m pytest -q
#
# With coverage:
#   python -m pytest Data -q --cov=Data --cov-report=term-missing
#
# Layout:
#   test_metrics.py      the science, as pure functions against golden vectors
#   test_endpoints.py    request validation and the response contract the
#                        frontend reads, driven by a query-routing fake cursor
#   test_db_endpoints.py the endpoints against real SQL and real Cartopy renders
#   fixture_db.py        builds the throwaway database those run on, seeded with
#                        a field whose answer is known by construction
#   conftest.py          stubs the psycopg2 pool so importing flask_api needs no
#                        DB, and provides the fixture-database fixtures
#
# Only test_db_endpoints.py needs PostgreSQL, and it skips itself when there is
# none (or with WEAVE_SKIP_DB_TESTS=1), so the rest still runs anywhere.

pytest>=8.0
pytest-cov>=5.0
